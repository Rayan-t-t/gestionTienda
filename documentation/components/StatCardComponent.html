import { ChangeDetectionStrategy, Component, input, output } from '@angular/core';
import { IonCard, IonCardContent, IonIcon } from '@ionic/angular';

/**
 * Tarjeta compacta para mostrar una métrica y su tendencia.
 */
@Component({
  selector: 'app-stat-card',
  standalone: true,
  imports: [IonCard, IonCardContent, IonIcon],
  templateUrl: './stat-card.component.html',
  styleUrl: './stat-card.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class StatCardComponent {
  readonly titulo = input.required<string>();
  readonly valor = input.required<string | number>();
  readonly icono = input.required<string>();
  readonly tendencia = input<string | null>(null);
  readonly pulsado = output<void>();
}

<ion-card>
  <button type="button" class="contenido" (click)="pulsado.emit()" [attr.aria-label]="titulo()">
    <ion-card-content>
      <ion-icon [name]="icono()" aria-hidden="true"></ion-icon>
      <span class="titulo">{{ titulo() }}</span>
      <strong class="valor">{{ valor() }}</strong>
      @if (tendencia(); as valorTendencia) {
        <span class="tendencia">{{ valorTendencia }}</span>
      }
    </ion-card-content>
  </button>
</ion-card>
